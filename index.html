function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Ambil data Reservasi var resSheet = ss.getSheetByName("reservations") || ss.getSheets()[0]; var resData = getSheetData(resSheet); // 2. Ambil data Transaksi (POS) var txSheet = ss.getSheetByName("transactions"); var txData = txSheet ? getSheetData(txSheet) : []; // 3. Ambil data Pengeluaran var expSheet = ss.getSheetByName("expenses"); var expData = expSheet ? getSheetData(expSheet) : []; var result = { reservations: resData, transactions: txData, expenses: expData }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // PENANGANAN TRANSAKSI POS BARU if (data.type === 'transaction') { var txSheet = ss.getSheetByName("transactions"); if (!txSheet) { txSheet = ss.insertSheet("transactions"); txSheet.appendRow(["id transaksi", "tanggal", "jam", "barber", "metode", "layanan", "total", "status"]); } txSheet.appendRow([ data.id, data.date, data.time, data.barberName, data.paymentMethod, data.items, data.total, data.status ]); } // PENANGANAN PENGELUARAN & KASBON else if (data.type === 'expense') { var expSheet = ss.getSheetByName("expenses"); if (!expSheet) { expSheet = ss.insertSheet("expenses"); expSheet.appendRow(["id pengeluaran", "tanggal", "keterangan", "metode", "jumlah"]); } expSheet.appendRow([ data.id, data.date, data.desc, data.paymentMethod, data.amount ]); } // UPDATE STATUS RESERVASI else if (data.type === 'update_reservation_status') { var resSheet = ss.getSheetByName("reservations") || ss.getSheets()[0]; resSheet.getRange(data.rowIndex + 1, 7).setValue(data.status); // Sesuaikan indeks kolom status } // UPDATE PELUNASAN KASBON else if (data.type === 'update_expense_amount') { var expSheet = ss.getSheetByName("expenses"); if (expSheet) { var values = expSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] == data.id) { expSheet.getRange(i + 1, 5).setValue(data.amount); // Kolom ke-5 adalah 'jumlah' break; } } } } // SIMPAN RESERVASI BARU PELANGGAN else { var resSheet = ss.getSheetByName("reservations") || ss.getSheets()[0]; resSheet.appendRow([ data.name, data.phone, data.barber, data.services, data.date, data.time, data.status, data.timestamp ]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // Helper untuk membaca Sheet menjadi Array of Objects function getSheetData(sheet) { var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0].map(function(h) { return String(h).toLowerCase().trim(); }); var data = []; for (var i = 1; i < values.length; i++) { var row = values[i]; var obj = { rowIndex: i }; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = row[j]; } data.push(obj); } return data; }
